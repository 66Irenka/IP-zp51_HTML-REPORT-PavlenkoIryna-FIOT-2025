Завдання №2.7

Умова завдання

Розробити клас Calculator, який підтримує ланцюжковий виклик методів (method chaining) для виконання арифметичних операцій. Реалізувати методи: number(), add(), subtract(), multiply(), divide() та getResult().

Програмний код


function task2_7() {
  class Calculator {
    constructor() {
      this.result = 0;
    }

    number(value) {
      this.result = value;
      return this;
    }

    add(value) {
      this.result += value;
      return this;
    }

    subtract(value) {
      this.result -= value;
      return this;
    }

    multiply(value) {
      this.result *= value;
      return this;
    }

    divide(value) {
      if (value === 0) {
        throw new Error("Ділення на нуль неможливе");
      }

      this.result /= value;
      return this;
    }

    getResult() {
      return this.result;
    }
  }

  const calc = new Calculator();

  const result = calc
    .number(10)
    .add(5)
    .subtract(3)
    .multiply(4)
    .divide(2)
    .getResult();

  console.log("Результат обчислення:", result);
}

Результат виконання

Після послідовного виконання операцій: 10 + 5 - 3 = 12, 12 * 4 = 48, 48 / 2 = 24. 

Результат обчислення: 24
